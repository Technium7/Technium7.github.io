\begin{rSection}{Experience}

\institution{Cornell University}
\role{Postbac research assistant, graduate research student}{Jun 2022 - Present}

\category{Robotics research}
\begin{experienceItems}
\item \textbf{Designed and implemented compressed-memory retrieval for VLA steering}. Manuscript in preparation.
\begin{details}
  \item Compressed the frozen SmolVLA VLM's 960-dimensional output (1.3 KB) into 64-dimensional keys (128 bytes) using a model trained with a soft dynamic time-warping contrastive loss based on action similarity.
  \item Initial tests showed significant gains on precision tasks: with a 300-episode memory (1.5 MB), direct rollout success increased from 33\% to 62\% across 100 held-out peg-insertion tasks.
\end{details}
\item \textbf{Built and deployed a tabletop robotic arm to automate heterogeneous precision operations in an optics laboratory}. Work submitted to ICRA 2027.
\begin{details}
  \item Developed and deployed robot perception, control, and motion software that enables operation in the existing lab environment with one additional top-view camera.
  \item Developed a multi-camera perception-and-control system with $\sim\!0.95$ mm localization uncertainty and a task-specific accuracy of $\leq\!0.2$ mm, enabling reliable automation of sensor manipulation, power measurement, and photonic-chip packaging.
\end{details}
\end{experienceItems}

\category{Quantum science research}
\begin{experienceItems}
\item \textbf{Co-optimized ion-trap junction electrodes and transport trajectories using a Bayesian neural network}. Manuscript in preparation.
\begin{details}
    \item Applied a BLIPS-style neural network and the cross-entropy method to optimize transport paths and junction designs that reduced the final phonon number by $\sim\!100\times$ for high-speed transport under noise.
    \item Built a quadratic-programming solver for dynamic multi-voltage control, reducing ion-transport complexity from $\mathcal{O}(nm)$ to $\mathcal{O}(n)$. Used the resulting voltage solutions to demonstrate ion transport on chips with multilayer integrated photonics for the first time.
\end{details}
\item \textbf{Demonstrated novel laser-cooling schemes for trapped ions} that reduced total cooling time by $>\!10\times$, addressing a bottleneck in industrial trapped-ion quantum computing systems in which cooling dominates operating time. Results led to two first-author papers.
\item \textbf{Designed and built a trapped-ion quantum computing system} as a founding student in the Photonics and Quantum Electronics lab.
\begin{details}
\item Built a system comprising multistage cryogenic vacuum chambers; DC and RF filtering and stabilization circuits; and free-space optics, fiber optics, and integrated photonics for addressing ions with narrow-linewidth UV--IR lasers. Developed FPGA-, DDS-, and DAC-based electrical and optical controls.
\item Developed and deployed Python packages to accelerate ion-trap chip design and automate system calibration, data collection, and in situ analysis. Several are now used by research groups at ETH and Duke.
\end{details}
\end{experienceItems}
